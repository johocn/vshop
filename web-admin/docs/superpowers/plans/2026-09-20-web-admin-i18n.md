> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 web-admin（uni-app Vue3 H5）搭建官方内置 i18n 基础设施，并抽取登录页、选店页两处作示范，验证「语言包 → 切换 → 即时生效」闭环。

**Tech Stack:** uni-app Vue3、TypeScript、JSON 语言包、H5（hash 路由，base `/guanli/`）。

### Task 3: 语言持久化与切换 store

**Files:**
- Create: `src/stores/localeStore.ts`

- [ ] **Step 1: 创建 `src/stores/localeStore.ts`**

```ts
import { defineStore } from 'pinia';
import { ref } from 'vue';
import { getLocale, setLocale } from '@dcloudio/uni-app';

const STORAGE_KEY = 'wa_locale';
const SUPPORTED = ['zh-Hans', 'en'];

export const useLocaleStore = defineStore('locale', () => {
  const locale = ref<string>(load());

  function load(): string {
    let v = '';
    try {
      v = uni.getStorageSync(STORAGE_KEY) as string;
    } catch (_e) { /* ignore */ }
    return SUPPORTED.includes(v) ? v : 'zh-Hans';
  }

  function apply(localeOrNull?: string | null): void {
    const target = localeOrNull ?? locale.value;
    try {
      if (SUPPORTED.includes(target)) setLocale(target);
      // setLocale 触发 vue3 自动重渲染；Storage 持久化
      uni.setStorageSync(STORAGE_KEY, target);
      locale.value = target;
    } catch (_e) {
      // 非法 locale 回退中文
      uni.setStorageSync(STORAGE_KEY, 'zh-Hans');
      locale.value = 'zh-Hans';
    }
  }

  // 校验当前 runtime locale，缺失时回退
  function ensure(): void {
    const cur = (() => { try { return getLocale(); } catch (_e) { return ''; } })();
    if (!SUPPORTED.includes(cur)) apply();
  }

  return { locale, apply, ensure };
});
```

### Task 8: 双语截图验证（登录页 zh/en 对照）

**Files:**
- Create: `scripts/i18n_verify.py`（复用既有 Playwright 截图框架）— 生成截图即可，不入手册正文必读

- [ ] **Step 1: 编写 `scripts/i18n_verify.py`**

```python
# -*- coding: utf-8 -*-
"""i18n 地基验证：登录页 zh-Hans / en 双语对照截图（视口 390x844 dpr=2）"""
import time
from playwright.sync_api import sync_playwright
BASE='https://e.joho.cn/guanli/'
SHOT=r'd:\zhao\vshop\web-admin\src\static\manual\shots'
def shoot(locale, name):
    with sync_playwright() as p:
        b=p.chromium.launch(); c=b.new_context(viewport={'width':390,'height':844},device_scale_factor=2)
        pg=c.new_page()
        pg.add_init_script(f"localStorage.setItem('wa_locale','{locale}')")
        pg.goto(BASE+'#/pages/login/index',wait_until='networkidle',timeout=30000)
        time.sleep(2)
        pg.screenshot(path=f'{SHOT}/{name}.png',full_page=False)
        b.close()
if __name__=='__main__':
    shoot('zh-Hans','i18n_login_zh')
    shoot('en','i18n_login_en')
    print('DONE')
```
