- [ ] **Step 1: 新建 `distribution-admin-shop.resolver.ts`**

```typescript
@Resolver()
export class DistributionAdminShopResolver {
    constructor(
        private distributionService: DistributionService,
        private commissionService: CommissionService,
        private withdrawalService: WithdrawalService,
        private customerService: CustomerService,
    ) {}

    @Query('distributors')
    @Allow(Permission.SuperAdmin, Permission.ReadCustomer)
    async distributors(@Ctx() ctx, @Args() options) {
        const list = await this.distributionService.findAll(ctx, options);
        // 附带 customerEmail 便于后台识别分销员
        const items = await Promise.all(list.items.map(async d => {
            const customer = d.customerId ? await this.customerService.findOne(ctx, d.customerId) : null;
            return { ...d, customerEmail: customer?.emailAddress ?? null };
        }));
        return { items, totalItems: list.totalItems };
    }

    @Query('commissionRecords') @Allow(Permission.SuperAdmin, Permission.ReadCustomer)
    async commissionRecords(@Ctx() ctx, @Args() options) { return this.commissionService.findAll(ctx, options); }

    @Query('withdrawalRequests') @Allow(Permission.SuperAdmin, Permission.ReadCustomer)
    async withdrawalRequests(@Ctx() ctx, @Args() options) { return this.withdrawalService.findAll(ctx, options); }

    @Mutation('approveDistributor') @Allow(Permission.SuperAdmin, Permission.UpdateCustomer)
    approveDistributor(@Ctx() ctx, @Args('id') id: ID) { return this.distributionService.approve(ctx, id); }

    @Mutation('freezeDistributor') @Allow(Permission.SuperAdmin, Permission.UpdateCustomer)
    freezeDistributor(@Ctx() ctx, @Args('id') id: ID) { return this.distributionService.freeze(ctx, id); }

    @Mutation('approveWithdrawal') @Allow(... )
    approveWithdrawal(@Ctx() ctx, @Args('id') id: ID) { return this.withdrawalService.approve(ctx, id); }

    @Mutation('rejectWithdrawal') @Allow(...)
    rejectWithdrawal(@Ctx() ctx, @Args('id') id: ID) { return this.withdrawalService.reject(ctx, id); }

    @Mutation('markWithdrawalPaid') @Allow(...)
    markWithdrawalPaid(@Ctx() ctx, @Args('id') id: ID) { return this.withdrawalService.markPaid(ctx, id); }
}
```

## 依赖与风险

- 金额单位为分，前后端换算需一致（后端 Int 分，前端 ÷100 显示）。提现表单前端传元、请求时转分。
- 分销后台操作挂 shop-api 用 `@Allow(SuperAdmin,…)` 门控，未授权用户调用会返回权限错误；前端后台页无入口，直接 URL 访问（沿用现有 admin 页模式），安全性依赖后端权限守卫。
- `distributors` 附带 `customerEmail` 需每行查一次 customer，列表量大时注意分页（默认仅取当页）。
- admin-api 里的旧 resolver 保留不影响（与 shop-api 并存，兼容既有后台）。
